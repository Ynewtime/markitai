1"""Machine-readable run result for ``markitai --json``. 

2 

3The CLI's four conversion paths already collect one :class:`Outcome` per work 

4item for ``--record-history``. This module renders that same collection as a 

5single JSON document on stdout, so a script or agent can read 

6``{source, status, output, error, cost_usd}`` without parsing Rich panels or 

7hunting for the batch report file. 

8 

9The module holds no process state: the CLI's own ``--json`` flag decides 

10whether the envelope is emitted, and every function here is pure. That keeps 

11one source of truth for "is JSON mode on" and makes the renderers trivial to 

12test. 

13 

14This module sits below :mod:`markitai.cli` and must never import it (the 

15import-layer contracts in the root ``pyproject.toml`` make that direction 

16forbidden); the CLI imports this module, never the other way round. 

17""" 

18 

19from __future__ import annotations 

20 

21import json 

22from typing import Any 

23 

24from markitai.runs.types import Outcome 

25 

26ENVELOPE_VERSION = "1.0" 

27 

28 

29def _item(outcome: Outcome) -> dict[str, Any]: 

30 """Render one Outcome as a stable, flat JSON object.""" 

31 return { 

32 "kind": outcome.kind, 

33 "source": outcome.source, 

34 "status": outcome.status, 

35 "output": str(outcome.output_path) if outcome.output_path else None, 

36 "error": outcome.error, 

37 "skip_reason": outcome.skip_reason, 

38 "images": outcome.images, 

39 "screenshots": outcome.screenshots, 

40 "cost_usd": round(outcome.cost_usd, 6), 

41 "duration_s": ( 

42 round(outcome.duration, 3) if outcome.duration is not None else None 

43 ), 

44 "cache_hit": outcome.cache_hit, 

45 "fetch_cache_hit": outcome.fetch_cache_hit, 

46 "llm_cache_hit": outcome.llm_cache_hit, 

47 "fetch_strategy": outcome.fetch_strategy, 

48 "source_file": outcome.source_file, 

49 "llm_usage": outcome.llm_usage, 

50 } 

51 

52 

53def build_envelope( 

54 outcomes: list[Outcome], *, error: str | None = None 

55) -> dict[str, Any]: 

56 """Build the top-level ``--json`` document from collected outcomes. 

57 

58 Args: 

59 outcomes: One Outcome per work item, in completion order. 

60 error: A run-level failure that produced no Outcome (a missing input 

61 path, an unsupported format, an interrupt). Without it, a run that 

62 died before its first item would report ``ok: true`` with an empty 

63 list while exiting non-zero. 

64 

65 Returns: 

66 ``{"version", "ok", "error", "items", "totals"}`` where ``ok`` is False 

67 when any item failed or a run-level error was reported (partial batch 

68 failure included). 

69 """ 

70 items = [_item(outcome) for outcome in outcomes] 

71 completed = sum(1 for i in items if i["status"] == "completed") 

72 failed = sum(1 for i in items if i["status"] == "failed") 

73 skipped = sum(1 for i in items if i["status"] == "skipped") 

74 return { 

75 "version": ENVELOPE_VERSION, 

76 "ok": failed == 0 and error is None, 

77 "error": error, 

78 "items": items, 

79 "totals": { 

80 "total": len(items), 

81 "completed": completed, 

82 "failed": failed, 

83 "skipped": skipped, 

84 "cost_usd": round(sum(i["cost_usd"] for i in items), 6), 

85 "duration_s": round( 

86 sum(i["duration_s"] or 0.0 for i in items), 

87 3, 

88 ), 

89 }, 

90 } 

91 

92 

93def render(outcomes: list[Outcome], *, error: str | None = None) -> str: 

94 """Render the envelope as pretty JSON with a trailing newline. 

95 

96 ``ensure_ascii`` stays on: a CJK source name written raw raises 

97 ``UnicodeEncodeError`` when stdout is an ASCII stream (``LC_ALL=C``, a 

98 Windows pipe), and escaped JSON is still valid UTF-8 JSON for ``jq``. 

99 """ 

100 return ( 

101 json.dumps(build_envelope(outcomes, error=error), indent=2, ensure_ascii=True) 

102 + "\n" 

103 ) 

104 

105

107 "ENVELOPE_VERSION", 

108 "build_envelope", 

109 "render",
