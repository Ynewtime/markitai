33# Raw-HTML markers for the document sniff below. ``matches_document`` can only 

34# succeed when the page carries a Substack preload payload or a substackcdn 

35# asset, so a page without either marker never needs to be parsed just to be 

36# rejected. Deliberately a superset of the real check: it only decides whether 

37# parsing is worth it, never whether the extractor matches. 

38_DOCUMENT_SNIFF_MARKERS = re.compile(r"substackcdn|_preloads", re.IGNORECASE) 

39 

40# Reused instance: ``matches_document`` is stateless, so constructing a fresh 

41# extractor per lookup only allocated garbage. 

42_SUBSTACK_EXTRACTOR = next( 

43 extractor 

44 for extractor in _EXTRACTORS 

45 if isinstance(extractor, SubstackNoteExtractor)

49def document_sniff_may_apply(html: str) -> bool: 

50 """Return whether *html* can possibly match a document-sniffing extractor. 

51 

52 Cheap raw-string pre-check that lets callers skip a full BeautifulSoup 

53 parse for the overwhelming majority of pages, which no extractor claims. 

54 

55 Args: 

56 html: Raw HTML source. 

57 

58 Returns: 

59 True when the document sniff is worth running. 

60 """ 

61 return _DOCUMENT_SNIFF_MARKERS.search(html) is not None
