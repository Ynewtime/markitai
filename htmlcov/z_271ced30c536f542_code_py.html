83def _strip_inline_numeric_gutters(root: Tag) -> None: 

84 """Drop numeric gutters from two-child line wrappers inside code. 

85 

86 Some viewers render each code line as a row whose first child is the 

87 line number and second the code (e.g. Chroma inline line numbers: 

88 ``<span style="display:flex"><span>1</span><span>code</span></span>``). 

89 Mirrors the two-child all-digits rule in upstream's 

90 ``extractStructuredText``; without it the text concatenates as 

91 ``"1p = 61"``. 

92 """ 

93 for el in root.select("pre span, pre div, code span, code div"): 

94 children = el.find_all(True, recursive=False) 

95 if len(children) != 2: 

96 continue 

97 if not children[0].get_text(strip=True).isdigit(): 

98 continue 

99 # A plain numeric literal also sits first in a two-token span 

100 # (``<span class="mi">42</span><span> + x</span>``); only rows that 

101 # announce themselves as line rows lose their first child. 

102 if _looks_like_line_row(el, children[0]): 

103 children[0].decompose()

111def _looks_like_line_row(row: Tag, gutter: Tag) -> bool:

244 """Extract language from element's class or data attributes. 

245 

246 Fuzzy pattern hits (e.g. ``CodeBlock-code``) only count when the 

247 detected token is a known language (defuddle's CODE_LANGUAGES check). 

248 """ 

249 # Check data attributes first 

250 for attr in ("data-lang", "data-language", "language"): 

251 val = el.get(attr) 

252 if val and isinstance(val, str): 

253 return val.strip().lower()

259 for pattern in _LANG_PATTERNS:

262 return match.group(1).lower()
